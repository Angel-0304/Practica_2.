config system interface
    edit "port3"
        set mode dhcp
        set defaultgw disable
        set allowaccess ping https ssh http
    next
end

#Hostname (GUI: System > Settings)
config system global
    set hostname "FG-Server"
end

#Interfaces (GUI: Network > Interfaces)
config system interface
    edit "port1"
        set alias "WAN"
        set mode static
        set ip 20.24.56.2 255.255.255.252
        set allowaccess ping
        set role wan
    next
    edit "port2"
        set mode static
        set ip 10.23.56.129 255.255.255.240
        set allowaccess ping
        set role lan
    next
end

#Ruta por defecto hacia el ISP (GUI: Network > Static Routes)
config router static
    edit 1
        set gateway 20.24.56.1
        set device "port1"
    next
end

# ============================================================
# OBJETOS (GUI: Policy & Objects > Addresses)
# ============================================================
config firewall address
    edit "Web-Server"
        set subnet 10.23.56.130 255.255.255.255
    next
    edit "POOL-VPN"
        set type iprange
        set start-ip 10.23.56.200
        set end-ip 10.23.56.210
    next
end

# ============================================================
# PUBLICACION DE LA WEB SIN VPN (GUI: Policy & Objects > Virtual IPs)
# (Solo se publica TCP 443; el puerto 22 NO queda expuesto)
# ============================================================
config firewall vip
    edit "VIP-WEB-HTTPS"
        set extip 20.24.56.2
        set mappedip "10.23.56.130"
        set extintf "port1"
        set portforward enable
        set extport 443
        set mappedport 443
    next
end

# ============================================================
# USUARIO Y GRUPO (GUI: User & Authentication)
# ============================================================
config user local
    edit "angel"
        set type password
        set passwd <PASSWORD>
    next
end

config user group
    edit "VPN-USUARIOS"
        set member "angel"
    next
end

# ============================================================
# SSL-VPN (intento inicial, DESCARTADO)
# La licencia de evaluacion (LENC) solo ofrece cifrados que los
# clientes actuales no soportan ("Cipher is (NONE)"). Se dejo
# escuchando en port4 (sin cable ni IP) para desactivarla.
# ============================================================
config vpn ssl settings
    set source-interface "port4"
    set port 10443
    set tunnel-ip-pools "SSLVPN_TUNNEL_ADDR1"
    set ssl-min-proto-ver tls1-0
end

# ============================================================
# VPN DE ACCESO REMOTO IPsec (GUI: VPN > IPsec Wizard > Remote Access
# ============================================================
config firewall address
    edit "VPN-REMOTO_range"
        set type iprange
        set start-ip 10.23.56.200
        set end-ip 10.23.56.210
        set comment "VPN: VPN-REMOTO (Created by VPN wizard)"
    next
end

#Phase 1
#(DES/MD5 y grupo DH 5, limitado por la licencia de evaluacion)
config vpn ipsec phase1-interface
    edit "VPN-REMOTO"
        set type dynamic
        set interface "port1"
        set ike-version 1
        set mode aggressive
        set peertype any
        set net-device disable
        set mode-cfg enable
        set proposal des-md5
        set dhgrp 5
        set keylife 86400
        set xauthtype auto
        set authusrgrp "VPN-USUARIOS"
        set ipv4-start-ip 10.23.56.200
        set ipv4-end-ip 10.23.56.210
        set ipv4-netmask 255.255.255.255
        set ipv4-split-include "Web-Server"
        set psksecret <PRE-SHARED-KEY>
        set comments "VPN: VPN-REMOTO (Created by VPN wizard)"
    next
end

#Phase 2
#(Selectores 0.0.0.0/0: normal en acceso remoto; la politica limita el acceso)
config vpn ipsec phase2-interface
    edit "VPN-REMOTO"
        set phase1name "VPN-REMOTO"
        set proposal des-md5
        set pfs enable
        set dhgrp 5
        set keylifeseconds 43200
    next
end

# ============================================================
# POLITICAS (GUI: Policy & Objects > Firewall Policy)
# ============================================================
config firewall policy
    edit 1
        set name "LAN-a-Internet"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "all"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set nat enable
    next
    edit 2
        set name "Internet-a-Web-HTTPS"
        set srcintf "port1"
        set dstintf "port2"
        set srcaddr "all"
        set dstaddr "VIP-WEB-HTTPS"
        set action accept
        set schedule "always"
        set service "HTTPS"
    next
    edit 3
        set name "VPN-a-Servidor-SSH"
        set srcintf "VPN-REMOTO"
        set dstintf "port2"
        set srcaddr "VPN-REMOTO_range"
        set dstaddr "Web-Server"
        set action accept
        set schedule "always"
        set service "SSH" "PING" "TRACEROUTE"
    next
end

PC-Usuario
=========================================================================

dhclient eth1
ip a
ip route

#Instalacion de paquetes (temporalmente por la red de Docker)
ip route replace default via 10.177.0.1 dev eth0
echo "nameserver 8.8.8.8" > /etc/resolv.conf
apt -o Acquire::ForceIPv4=true update
apt -o Acquire::ForceIPv4=true install -y strongswan libcharon-extra-plugins libstrongswan-extra-plugins traceroute openssh-client curl
ip route replace default via 10.23.56.1 dev eth1    # devolver ruta al R-USER

#Cliente VPN: /etc/ipsec.conf
config setup
    charondebug="ike 1, knl 1, cfg 1"

conn fortigate
    keyexchange=ikev1
    aggressive=yes
    ike=des-md5-modp1536!
    esp=des-md5-modp1536!
    left=%defaultroute
    leftid=@angel
    leftauth=psk
    leftauth2=xauth
    xauth_identity=angel
    leftsourceip=%config
    right=20.24.56.2
    rightid=%any
    rightauth=psk
    rightsubnet=10.23.56.130/32
    auto=add

#Credenciales: /etc/ipsec.secrets  (chmod 600)
: PSK "<PRE-SHARED-KEY>"
angel : XAUTH "<PASSWORD>"

#Pruebas SIN VPN
curl -k https://20.24.56.2                      # Muestra la pagina (web publica)
ssh -o ConnectTimeout=5 angel@20.24.56.2        # Connection timed out
ssh -o ConnectTimeout=5 angel@10.23.56.130      # No route to host

#Conectar la VPN
ipsec start
ipsec up fortigate
# XAuth authentication of 'angel' (myself) successful
# installing new virtual IP 10.23.56.200
# connection 'fortigate' established successfully

#Pruebas CON VPN
ping -c 4 10.23.56.130                          # TTL=63
traceroute -n 10.23.56.130
#  1  * * *        (FG-Server: su respuesta ICMP queda fuera del selector)
#  2  10.23.56.130
ssh angel@10.23.56.130                          # Entra al servidor

#Desconectar la VPN
ipsec down fortigate
ipsec status                                    # 0 up, 0 connecting
curl -k https://20.24.56.2                      # La web sigue funcionando
ssh -o ConnectTimeout=5 angel@10.23.56.130      # Vuelve a fallar
